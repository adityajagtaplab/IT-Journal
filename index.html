Sum & product of digits (onblur)

HTML + JS 
<html>
<head>
<title>JS program - sum and product of digits</title>
<script>
function calc_digits() {
var n, d1, d2, d3, sum, prod;
n = parseInt(document.form1.num_box.value);
d1 = Math.floor(n / 100);
d2 = Math.floor((n / 10) % 10);
d3 = n % 10;
sum = d1 + d2 + d3;
prod = d1 * d2 * d3;
document.getElementById("disp").innerHTML = "Sum of digits - " + sum + "<br>Product of digits - " + prod;
}
</script>
</head>
<body>
<h1>JS program to find sum and product of digits of a 3 digit number</h1>
<form name="form1">
Enter a 3 digit number:
<input type="text" name="num_box" onblur="calc_digits()">
</form>
<br>
<p id="disp"> </p>
</body>
